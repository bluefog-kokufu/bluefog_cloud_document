function loadDb(){
  try{const s=localStorage.getItem(LS_KEY); if(s){db=JSON.parse(s);}}catch(e){}
  if(!db)db=JSON.parse(JSON.stringify(DEFAULT_DB));
  /* migration for saved data from older versions */
  if(!db.paymentNotices)db.paymentNotices=JSON.parse(JSON.stringify(DEFAULT_DB.paymentNotices));
  delete db.masters;
  if(!db.ledger)db.ledger=JSON.parse(JSON.stringify(DEFAULT_DB.ledger));
  /* 旧バージョンの行データ(単式の総勘定元帳形式/中間版の仕訳帳形式)を仕訳帳形式に移行 */
  (db.ledger.rows||[]).forEach(r=>{
    if(r.year==null){
      const y=(r.date||db.ledger.date||today()).slice(0,4);
      r.year=y;r.m=r.m||(r.date?String(Number(r.date.slice(5,7))):"");r.d=r.d||(r.date?String(Number(r.date.slice(8,10))):"");
      r.drAcct=r.drAcct||"";r.drAmt=r.drAmt||0;r.crAcct=r.crAcct||"";r.crAmt=r.crAmt||0;
      delete r.date;
    }
    /* 旧・単式(acct/dr/cr)のみのデータは借方/貸方に振り分ける */
    if(r.acct!=null){
      if(Number(r.dr)>0&&!r.drAcct){r.drAcct=r.acct;r.drAmt=Number(r.dr);}
      if(Number(r.cr)>0&&!r.crAcct){r.crAcct=r.acct;r.crAmt=Number(r.cr);}
      delete r.acct;delete r.dr;delete r.cr;
    }
    r.no=r.no||"";r.note=r.note||"";r.page=r.page||"";
  });
  (db.sales||[]).forEach(s=>{
    if(!s.items||!s.items.length) s.items=[{name:s.memo||"商品売上", amount:Number(s.amount)||0, rate:db.settings.taxRate}];
  });
  (db.paymentNotices||[]).forEach(n=>{
    (n.items||[]).forEach(it=>{ if(it.tax==="外税")it.tax="10%"; });
  });
  (db.customers||[]).forEach(c=>{
    if(c.custType==null){
      c.custType="両方で使用する";
      c.zip=c.zip||"";
      c.pref=c.pref||"";
      c.addr1=c.addr1||c.addr||"";
      c.addr2=c.addr2||"";
      c.mobile=c.mobile||"";
      c.fax=c.fax||"";
      c.url=c.url||"";
    }
  });
}

async function custZipLookup(){
  const zipEl=$("#cf_zip"), btn=$("#cf_zipbtn");
  const zip=(zipEl.value||"").replace(/[^0-9]/g,"");
  if(zip.length!==7){alert("郵便番号は7桁の数字で入力してください(例: 1000001)。");return;}
  const hit=ZIP_LOOKUP[zip];
  const origLabel=btn.textContent;
  btn.textContent="検索中…";btn.disabled=true;
  try{
    const res=await fetch(`https://zipcloud.ibsnet.co.jp/api/search?zipcode=${zip}`);
    if(!res.ok)throw new Error("http "+res.status);
    const data=await res.json();
    if(data.status===200 && data.results && data.results.length){
      const r=data.results[0];
      $("#cf_pref").value=r.address1;
      $("#cf_addr1").value=`${r.address2}${r.address3}`;
      zipEl.value=zip.slice(0,3)+"-"+zip.slice(3);
    }else if(hit){
      $("#cf_pref").value=hit.pref;$("#cf_addr1").value=hit.addr1;
    }else{
      alert("該当する住所が見つかりませんでした。郵便番号をご確認のうえ、都道府県・住所は直接入力してください。");
    }
  }catch(e){
    if(hit){
      $("#cf_pref").value=hit.pref;$("#cf_addr1").value=hit.addr1;
      alert("住所検索API(zipcloud)に接続できなかったため、内蔵データから入力しました。");
    }else{
      alert("住所検索に失敗しました。ネットワーク接続をご確認いただくか、都道府県・住所は直接入力してください。");
    }
  }finally{
    btn.textContent=origLabel;btn.disabled=false;
  }
}
function calcTaxAt(amount, rate){
  const r=Number(rate)/100, raw=(Number(amount)||0)*r, m=db.settings.rounding;
  return m==="ceil"?Math.ceil(raw):m==="round"?Math.round(raw):Math.floor(raw);
}
function calcTax(amount){ return calcTaxAt(amount, db.settings.taxRate); }
function addDays(dateStr, n){
  const d=new Date(dateStr+"T00:00:00"); d.setDate(d.getDate()+n);
  const p=x=>String(x).padStart(2,"0");
  return `${d.getFullYear()}-${p(d.getMonth()+1)}-${p(d.getDate())}`;
}
function csvDownload(filename, rows){
  const csv=rows.map(r=>r.map(c=>{c=String(c==null?"":c);return /[",\n]/.test(c)?'"'+c.replaceAll('"','""')+'"':c;}).join(",")).join("\r\n");
  const blob=new Blob(["﻿"+csv],{type:"text/csv"});
  const a=document.createElement("a");a.href=URL.createObjectURL(blob);a.download=filename;a.click();
  setTimeout(()=>URL.revokeObjectURL(a.href),3000);
}

  $("#"+prefix+"CsvFile").addEventListener("change",e=>{
    const f=e.target.files[0];if(!f)return;

    };
    r.readAsText(f,"UTF-8");
  });
}

function custCsvImport(rows){
  let added=0,updated=0;
  rows.forEach((x,i)=>{
    if(i===0&&x[0]===CUST_CSV_HEAD[0])return;
    const name=String(x[0]||"").trim();
    if(!name)return;
    const data={name,custType:["受注取引管理","発注取引管理","両方で使用する"].includes(x[1])?x[1]:"両方で使用する",
      zip:x[2]||"",pref:x[3]||"",addr1:x[4]||"",addr2:x[5]||"",tel:x[6]||"",mobile:x[7]||"",fax:x[8]||"",url:x[9]||"",
      person:x[10]||"",email:x[11]||"",memo:x[12]||""};
    const ex=db.customers.find(c=>c.name===name);
    if(ex){Object.assign(ex,data);updated++;}

  });

}

/* 適格請求書のルール: 端数処理(切り捨て/四捨五入/切り上げ)は「税率ごとに1回」。
   明細行ごとに計算して合算すると税額がずれるため、同一税率の税抜金額を
   先に合算してから消費税額を1回だけ算出する。 */
function groupByRate(items){
  const g={};
  (items||[]).forEach(it=>{
    const r=Number(it.rate)||0;
    if(!g[r]) g[r]={rate:r, sub:0};
    g[r].sub+=Number(it.amount)||0;
  });
  Object.values(g).forEach(v=>{ v.tax=calcTaxAt(v.sub, v.rate); });
  return g;
}
function saleTotals(items){
  const g=groupByRate(items);
  let sub=0,tax=0;
  Object.values(g).forEach(v=>{sub+=v.sub;tax+=v.tax;});
  return {sub,tax,total:sub+tax,groups:g};
}

function purCsvImport(rows){
  let added=0;
  rows.forEach((x,i)=>{
    // ヘッダー行(エクスポート形式/テンプレート形式の両方)をスキップ
    if(i===0&&(x[0]===PUR_CSV_HEAD[0]||x[0]==="取引年月日"))return;
    // エクスポート形式(No列あり: 日付は2列目)とテンプレート形式(No列なし: 日付は1列目)を自動判別
    const b=/^\d{4}-\d{2}-\d{2}$/.test(String(x[0]||"").trim())?0:1;
    const date=x[b]||today(), name=x[b+1], method=x[b+2]||"現金";
    if(!String(name||"").trim())return;
    const amount=Number(x[b+3])||0;
    const taxCell=String(x[b+4]==null?"":x[b+4]).trim();
    const tax=taxCell!==""&&!isNaN(Number(taxCell))?Number(taxCell):calcTax(amount);
    const status=["未払い","支払い済"].includes(x[b+5])?x[b+5]:"未払い";
    db.purchases.push({id:uid().slice(0,9),date,custId:custIdByName(name),method,amount,tax,status,memo:x[b+6]||"",files:{},up:today()});
    added++;
  });
  save();alert(`インポート完了: ${added}件の取引を登録しました`);show("purchases");
}

function ledgerSortKey(r){
  return `${String(r.year||"0").padStart(4,"0")}-${String(r.m||"0").padStart(2,"0")}-${String(r.d||"0").padStart(2,"0")}`;
}
function ledgerEntries(acct){
  const es=[];
  db.ledger.rows.forEach(r=>{
    if(r.drAcct===acct&&Number(r.drAmt))es.push({no:r.no,year:r.year,m:r.m,d:r.d,note:r.note,other:r.crAcct||"諸口",dr:Number(r.drAmt),cr:0});
    if(r.crAcct===acct&&Number(r.crAmt))es.push({no:r.no,year:r.year,m:r.m,d:r.d,note:r.note,other:r.drAcct||"諸口",dr:0,cr:Number(r.crAmt)});
  });
  return es.sort((a,b)=>ledgerSortKey(a).localeCompare(ledgerSortKey(b)));
}

function noticeTaxRate(tax){
  if(tax==="10%")return 10;
  if(tax==="8%"||tax==="8%軽減税率")return 8;
  return 0;
}
function noticeCalc(n){
  let sub=0,tax=0;
  (n.items||[]).forEach(it=>{
    const a=(Number(it.price)||0)*(Number(it.qty)||0);
    sub+=a; tax+=calcTaxAt(a,noticeTaxRate(it.tax));
  });
  return {sub,tax,total:sub+tax};
}

      </div>
    </div>`;
  $("#nf_cust").addEventListener("change",e=>payDraft.custId=e.target.value);
  $("#nf_date").addEventListener("change",e=>payDraft.payDate=e.target.value);
  $("#nf_title").addEventListener("input",e=>payDraft.title=e.target.value);
  /* 単価/数量/消費税/日付/品目/単位の変更では画面全体を再描画せず、
     金額セルと合計欄だけを直接書き換える(数値入力の上下矢印クリックで画面が先頭へ飛ぶのを防ぐため) */
  const payItemsChange=e=>{
    const i=e.target.dataset.i;if(i==null)return;
    const k=e.target.dataset.k;
    payDraft.items[i][k]=(k==="price"||k==="qty")?Number(e.target.value)||0:e.target.value;
    const amtCell=document.getElementById("payAmt"+i);
    if(amtCell)amtCell.textContent=num((Number(payDraft.items[i].price)||0)*(Number(payDraft.items[i].qty)||0));
    const c=noticeCalc(payDraft);
    if($("#pf_sub"))$("#pf_sub").textContent=yen(c.sub);
    if($("#pf_tax"))$("#pf_tax").textContent=yen(c.tax);
    if($("#pf_total"))$("#pf_total").textContent=yen(c.total);
  };
  $("#payItems").addEventListener("input",payItemsChange);
  $("#payItems").addEventListener("change",payItemsChange);
};

function parseCsv(text){
  const out=[];let row=[],cur="",q=false;
  text=String(text).replace(/^﻿/,"");
  for(let i=0;i<text.length;i++){
    const ch=text[i];
    if(q){
      if(ch==='"'){if(text[i+1]==='"'){cur+='"';i++;}else q=false;}
      else cur+=ch;
    }else{
      if(ch==='"')q=true;
      else if(ch===","){row.push(cur);cur="";}
      else if(ch==="\n"||ch==="\r"){if(ch==="\r"&&text[i+1]==="\n")i++;row.push(cur);out.push(row);row=[];cur="";}
      else cur+=ch;
    }
  }
  if(cur!==""||row.length){row.push(cur);out.push(row);}
  return out;
}
